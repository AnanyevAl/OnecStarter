```python
"""Поиск со звёздочкой (T-26, спека 2026-09-29-search-wildcard-design.md, §2).

`*` — «любые символы», всё остальное ищется буквально: в именах баз обычны
`.`, `(`, `+`, и регулярные выражения давали бы ошибки разбора и ложные
совпадения. Модуль `re` не используется намеренно.
"""  # noqa: RUF002


def matches_pattern(text: str, query: str) -> bool:
    """Совпадает ли `text` с запросом: части между `*` ищутся по порядку, без учёта регистра.

    Пустой запрос и запрос из одних `*` совпадают со всем. Совпадения частей
    не перекрываются; шаблон не привязан к началу и концу текста. Запрос без
    `*` — ровно прежняя проверка подстроки (`casefold`, [Ф] T-05.3).
    """  # noqa: RUF002
    parts = [part for part in query.strip().casefold().split("*") if part]
    folded = text.casefold()
    position = 0
    for part in parts:
        found = folded.find(part, position)
        if found < 0:
            return False
        position = found + len(part)
    return True
```

```python
def filter_rows(rows: Sequence[Row], query: str) -> list[Row]:
    """Отфильтровать лес по имени: подстрока или шаблон со звёздочкой (T-26).

    Правило совпадения — `domain.search.matches_pattern`: без учёта регистра
    ([Ф] T-05.3: платформа сравнивает имена баз регистронезависимо), `*` —
    любые символы. Совпавший узел остаётся со всем поддеревом; предок
    совпавшего — с отфильтрованными потомками. NOTE-строки сами не совпадают
    никогда и выживают только в поддереве совпавшего узла.
    """  # noqa: RUF002
    if not query.strip():
        return list(rows)
    kept: list[Row] = []
    for row in rows:
        if row.kind is not RowKind.NOTE and matches_pattern(row.label, query):
            kept.append(row)
            continue
        children = filter_rows(row.children, query)
        if children:
            kept.append(replace(row, children=tuple(children)))
    return kept
```

```python
class SearchField(QLineEdit):
    """Поле поиска раздела: подсказка с хоткеем и свой крестик очистки.

    `list_navigation` (T-35): Enter и Tab подают `list_requested` — раздел
    переводит фокус в свой список. Выключено по умолчанию: поле стоит и
    в диалоге «Цветовая схема…», где Tab обязан работать как обычно.
    """  # noqa: RUF002

    list_requested = Signal()
    # Атрибут класса, а не только экземпляра: `event()` может прийти во время
    # `super().__init__`, раньше присваивания в конструкторе.  # noqa: RUF003
    _list_navigation = False

    def __init__(
        self,
        hint: str,
        palette: Palette,
        parent: QWidget | None = None,
        *,
        list_navigation: bool = False,
    ) -> None:
        super().__init__(parent)
        self._list_navigation = list_navigation
        self.setPlaceholderText(hint + HOTKEY_SUFFIX)
        self._clear = QAction("Очистить", self)
        self._clear.setIcon(clear_icon(palette))
        self._clear.setVisible(False)
        self._clear.triggered.connect(self.clear)
        self.addAction(self._clear, QLineEdit.ActionPosition.TrailingPosition)
        self.textChanged.connect(lambda text: self._clear.setVisible(bool(text)))
        if list_navigation:
            self.returnPressed.connect(self.list_requested.emit)

    def event(self, event: QEvent) -> bool:
        # Tab Qt разбирает в QWidget.event (смена фокуса) раньше keyPressEvent,
        # поэтому перехват — здесь. Shift+Tab не трогаем.
        if (
            self._list_navigation
            and event.type() == QEvent.Type.KeyPress
            and isinstance(event, QKeyEvent)
            and event.key() == Qt.Key.Key_Tab
            and event.modifiers() == Qt.KeyboardModifier.NoModifier
        ):
            self.list_requested.emit()
            return True
        return super().event(event)
```

```python
    def _select_in_tree(self, key: str) -> bool:
        """Сделать текущей строку записи в ветке файла (T-36); `False` — не видна.

        Ветки верхнего уровня вида SECTION («Избранное», «Недавние», «Общие
        списки») пропускаются: запись выделяется там, где она лежит в файле.
        Не найдена — поиск её скрыл; текущая строка остаётся той, что вернул
        `_restore_current`, поиск не очищается (решение заказчика 20.10.2026).
        """  # noqa: RUF002
        model = self._tree.model()

        def walk(parent: QModelIndex) -> QModelIndex | None:
            for row in range(model.rowCount(parent)):
                index = model.index(row, 0, parent)
                if not parent.isValid() and index.data(KIND_ROLE) == RowKind.SECTION.value:
                    continue
                if index.data(KIND_ROLE) == RowKind.BASE.value and index.data(KEY_ROLE) == key:
                    return index
                found = walk(index)
                if found is not None:
                    return found
            return None

        found = walk(QModelIndex())
        if found is None:
            return False
        self._focus_index(found)
        return True
```

```python
def known_servers(items: Sequence[InfobaseItem]) -> list[str]:
    """Варианты поля «Сервер» (T-29): различающиеся `Srvr` серверных записей.

    Источник — все записи раздела «Базы», общие списки тоже (решение заказчика
    20.10.2026). Значение — строка целиком: список кластера через запятую
    остаётся одним вариантом. Дубли — только точные: как платформа сравнивает
    `Srvr` без учёта регистра, не измерено ([?]). Порядок — по алфавиту без
    учёта регистра, при равенстве — по исходной строке.
    """  # noqa: RUF002
    found: set[str] = set()
    for item in items:
        if item.is_group or item.kind is not ConnectKind.SERVER or not item.connect:
            continue
        value = find_fragment(parse_connect(item.connect), "Srvr")
        if value and value.strip():
            found.add(value)
    return sorted(found, key=lambda value: (value.casefold(), value))
```
